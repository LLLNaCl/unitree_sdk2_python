**通信模块示例**:
```python
import paho.mqtt.client as mqtt
import json
import ssl
from dotenv import load_dotenv
import os

load_dotenv()

class MQTTClient:
    def __init__(self, device_id):
        self.device_id = device_id
        self.client = mqtt.Client(client_id=device_id, protocol=mqtt.MQTTv5)
        self.client.username_pw_set(
            os.getenv("MQTT_USERNAME"), 
            os.getenv("MQTT_PASSWORD")
        )
        self.client.tls_set(cert_reqs=ssl.CERT_REQUIRED)
        self.client.on_connect = self._on_connect
        self.client.on_message = self._on_message
        
        # 主题订阅
        self.command_topic = f"device/{device_id}/command"
        self.task_topic = f"device/{device_id}/task"
        self.config_topic = f"device/{device_id}/config"
        
        # 发布主题
        self.status_topic = f"device/{device_id}/status"
        self.telemetry_topic = f"device/{device_id}/telemetry"
        self.task_result_topic = f"device/{device_id}/task/result"
        self.event_topic = f"device/{device_id}/event"
        
        # 消息回调函数
        self.command_callback = None
        self.task_callback = None
        self.config_callback = None
        
    def connect(self):
        """连接到MQTT服务器"""
        self.client.connect(
            os.getenv("MQTT_BROKER_HOST"),
            int(os.getenv("MQTT_BROKER_PORT")),
            keepalive=60
        )
        self.client.loop_start()
        
    def disconnect(self):
        """断开MQTT连接"""
        self.client.loop_stop()
        self.client.disconnect()
        
    def _on_connect(self, client, userdata, flags, rc, properties=None):
        """连接回调函数"""
        if rc == 0:
            print("Connected to MQTT broker successfully")
            # 订阅相关主题
            self.client.subscribe(self.command_topic)
            self.client.subscribe(self.task_topic)
            self.client.subscribe(self.config_topic)
        else:
            print(f"Failed to connect, return code {rc}")
            
    def _on_message(self, client, userdata, msg):
        """消息接收回调函数"""
        try:
            payload = json.loads(msg.payload.decode())
            print(f"Received message on {msg.topic}: {payload}")
            
            if msg.topic == self.command_topic and self.command_callback:
                self.command_callback(payload)
            elif msg.topic == self.task_topic and self.task_callback:
                self.task_callback(payload)
            elif msg.topic == self.config_topic and self.config_callback:
                self.config_callback(payload)
                
        except json.JSONDecodeError:
            print(f"Failed to decode JSON message: {msg.payload}")
            
    def publish_status(self, status_data):
        """发布设备状态"""
        self.client.publish(
            self.status_topic,
            json.dumps(status_data),
            qos=1,
            retain=True
        )
        
    def publish_telemetry(self, telemetry_data):
        """发布遥测数据"""
        self.client.publish(
            self.telemetry_topic,
            json.dumps(telemetry_data),
            qos=0
        )
        
    def publish_task_result(self, result_data):
        """发布任务执行结果"""
        self.client.publish(
            self.task_result_topic,
            json.dumps(result_data),
            qos=1
        )
        
    def publish_event(self, event_data):
        """发布设备事件"""
        self.client.publish(
            self.event_topic,
            json.dumps(event_data),
            qos=1
        )
```

**视频流模块示例**:
```python
import subprocess
import threading
import time
from dotenv import load_dotenv
import os

load_dotenv()

class VideoStreamer:
    def __init__(self):
        self.process = None
        self.running = False
        self.rtsp_url = os.getenv("RTSP_STREAM_URL")
        
    def start_stream(self):
        """启动视频流推流"""
        if self.running:
            print("Stream is already running")
            return
            
        # FFmpeg推流命令
        cmd = [
            "ffmpeg",
            "-f", "v4l2",
            "-i", "/dev/video0",
            "-s", "640x360",
            "-c:v", "libx264",
            "-preset", "ultrafast",
            "-f", "rtsp",
            self.rtsp_url
        ]
        
        try:
            self.process = subprocess.Popen(
                cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE
            )
            self.running = True
            print("Video stream started")
            
            # 启动错误监控线程
            threading.Thread(target=self._monitor_errors, daemon=True).start()
            
        except Exception as e:
            print(f"Failed to start video stream: {e}")
            
    def stop_stream(self):
        """停止视频流推流"""
        if not self.running:
            print("Stream is not running")
            return
            
        try:
            self.process.terminate()
            self.process.wait(timeout=5)
            self.running = False
            print("Video stream stopped")
        except Exception as e:
            print(f"Failed to stop video stream: {e}")
            
    def _monitor_errors(self):
        """监控FFmpeg错误输出"""
        while self.running:
            if self.process.poll() is not None:
                # 进程已退出
                stderr = self.process.stderr.read().decode()
                print(f"FFmpeg process exited with error: {stderr}")
                self.running = False
                # 尝试重启
                time.sleep(5)
                self.start_stream()
                break
                
            time.sleep(1)
```

### 5.3 主程序集成

```python
from communication import MQTTClient
from video_streamer import VideoStreamer
from device_manager import DeviceManager
from task_manager import TaskManager
from motion_controller import MotionController
from perception import PerceptionModule
from map_manager import MapManager
from dotenv import load_dotenv
import os
import time
import json

load_dotenv()

class RobotController:
    def __init__(self):
        self.device_id = os.getenv("DEVICE_ID")
        
        # 初始化各个模块
        self.mqtt_client = MQTTClient(self.device_id)
        self.video_streamer = VideoStreamer()
        self.device_manager = DeviceManager()
        self.task_manager = TaskManager()
        self.motion_controller = MotionController()
        self.perception = PerceptionModule()
        self.map_manager = MapManager()
        
        # 设置MQTT回调
        self.mqtt_client.command_callback = self._handle_command
        self.mqtt_client.task_callback = self._handle_task
        self.mqtt_client.config_callback = self._handle_config
        
        # 启动状态上报线程
        self.status_thread = threading.Thread(target=self._report_status, daemon=True)
        self.running = False
        
    def start(self):
        """启动机器人控制器"""
        try:
            self.running = True
            
            # 连接MQTT
            self.mqtt_client.connect()
            
            # 启动视频流
            self.video_streamer.start_stream()
            
            # 启动状态上报
            self.status_thread.start()
            
            print(f"Robot controller started for device {self.device_id}")
            
            # 注册设备
            self._register_device()
            
        except Exception as e:
            print(f"Failed to start robot controller: {e}")
            self.stop()
            
    def stop(self):
        """停止机器人控制器"""
        self.running = False
        
        # 停止视频流
        self.video_streamer.stop_stream()
        
        # 断开MQTT连接
        self.mqtt_client.disconnect()
        
        print("Robot controller stopped")
        
    def _register_device(self):
        """注册设备到云端"""
        hardware_info = self.device_manager.get_hardware_info()
        registration_result = self.device_manager.register_device(
            self.device_id, hardware_info
        )
        
        if registration_result:
            self.mqtt_client.publish_event({
                "event_type": "device_registered",
                "timestamp": int(time.time()),
                "status": "success",
                "device_info": hardware_info
            })
        else:
            self.mqtt_client.publish_event({
                "event_type": "device_registration_failed",
                "timestamp": int(time.time()),
                "status": "failed"
            })
            
    def _report_status(self):
        """定期上报设备状态"""
        while self.running:
            try:
                status_data = self.device_manager.get_device_status()
                self.mqtt_client.publish_status(status_data)
                
                # 上报遥测数据
                telemetry_data = self.perception.get_telemetry_data()
                self.mqtt_client.publish_telemetry(telemetry_data)
                
                # 上报任务状态
                task_status = self.task_manager.get_current_task_status()
                if task_status:
                    self.mqtt_client.publish_task_result({
                        "task_id": task_status["task_id"],
                        "status": task_status["status"],
                        "progress": task_status["progress"],
                        "timestamp": int(time.time())
                    })
                    
            except Exception as e:
                print(f"Error reporting status: {e}")
                
            # 每5秒上报一次
            time.sleep(5)
            
    def _handle_command(self, command):
        """处理控制命令"""
        try:
            command_id = command.get("message_id", f"cmd-{int(time.time())}")
            command_type = command.get("command")
            parameters = command.get("parameters", {})
            
            print(f"Handling command: {command_type} with parameters: {parameters}")
            
            # 执行命令
            if command_type == "move":
                result = self.motion_controller.move(**parameters)
            elif command_type == "rotate":
                result = self.motion_controller.rotate(** parameters)
            elif command_type == "start_stream":
                self.video_streamer.start_stream()
                result = {"status": "success", "message": "Stream started"}
            elif command_type == "stop_stream":
                self.video_streamer.stop_stream()
                result = {"status": "success", "message": "Stream stopped"}
            elif command_type == "get_sensor_data":
                sensor_type = parameters.get("sensor_type")
                result = self.perception.get_sensor_data(sensor_type)
            else:
                result = {"status": "error", "message": f"Unknown command: {command_type}"}
                
            # 回复命令结果
            self.mqtt_client.publish_task_result({
                "message_id": command_id,
                "command": command_type,
                "status": result.get("status", "error"),
                "result": result,
                "timestamp": int(time.time())
            })
            
        except Exception as e:
            print(f"Error handling command: {e}")
            self.mqtt_client.publish_task_result({
                "message_id": command.get("message_id"),
                "command": command.get("command"),
                "status": "error",
                "error": str(e),
                "timestamp": int(time.time())
            })
            
    def _handle_task(self, task):
        """处理任务分配"""
        try:
            task_id = task.get("task_id")
            task_type = task.get("task_type")
            parameters = task.get("parameters", {})
            
            print(f"Handling task: {task_type} (ID: {task_id}) with parameters: {parameters}")
            
            # 添加任务到队列
            task_result = self.task_manager.add_task(
                task_id, task_type, parameters, 
                priority=task.get("priority", 1),
                deadline=task.get("deadline")
            )
            
            # 回复任务接收状态
            self.mqtt_client.publish_task_result({
                "task_id": task_id,
                "status": "accepted" if task_result else "rejected",
                "timestamp": int(time.time())
            })
            
        except Exception as e:
            print(f"Error handling task: {e}")
            self.mqtt_client.publish_task_result({
                "task_id": task.get("task_id"),
                "status": "error",
                "error": str(e),
                "timestamp": int(time.time())
            })
            
    def _handle_config(self, config):
        """处理配置更新"""
        try:
            config_type = config.get("config_type")
            config_data = config.get("config_data")
            
            print(f"Handling config update: {config_type}")
            
            # 应用配置
            if config_type == "network":
                result = self.device_manager.update_network_config(config_data)
            elif config_type == "video":
                result = self.video_streamer.update_config(config_data)
            elif config_type == "motion":
                result = self.motion_controller.update_config(config_data)
            else:
                result = {"status": "error", "message": f"Unknown config type: {config_type}"}
                
            # 回复配置更新结果
            self.mqtt_client.publish_event({
                "event_type": "config_updated",
                "config_type": config_type,
                "status": result.get("status", "error"),
                "timestamp": int(time.time())
            })
            
        except Exception as e:
            print(f"Error handling config: {e}")
            self.mqtt_client.publish_event({
                "event_type": "config_update_failed",
                "config_type": config.get("config_type"),
                "error": str(e),
                "timestamp": int(time.time())
            })

if __name__ == "__main__":
    controller = RobotController()
    try:
        controller.start()
        # 保持主程序运行
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        print("Shutting down...")
        controller.stop()
```

```python
# network_manager.py
import subprocess
import time
import re

class NetworkManager:
    def __init__(self):
        self.wifi_connected = False
        self.modem_connected = False
        self.current_interface = None
        
    def check_network_status(self):
        """检查网络连接状态"""
        try:
            # 检查WiFi状态
            wifi_result = subprocess.run(
                ["iwgetid"], capture_output=True, text=True
            )
            self.wifi_connected = wifi_result.returncode == 0
            
            # 检查4G模块状态
            modem_result = subprocess.run(
                ["mmcli", "-L"], capture_output=True, text=True
            )
            self.modem_connected = "modem" in modem_result.stdout
            
            # 确定当前使用的网络接口
            if self.wifi_connected:
                self.current_interface = "wifi"
            elif self.modem_connected:
                self.current_interface = "4g"
            else:
                self.current_interface = None
                
            return {
                "wifi_connected": self.wifi_connected,
                "modem_connected": self.modem_connected,
                "current_interface": self.current_interface,
                "signal_strength": self.get_signal_strength()
            }
            
        except Exception as e:
            print(f"Error checking network status: {e}")
            return {"error": str(e)}
            
    def get_signal_strength(self):
        """获取信号强度"""
        if self.current_interface == "wifi":
            try:
                result = subprocess.run(
                    ["iwconfig", "wlan0"], capture_output=True, text=True
                )
                signal_match = re.search(r"Signal level=(-\d+) dBm", result.stdout)
                if signal_match:
                    return int(signal_match.group(1))
            except:
                pass
                
        elif self.current_interface == "4g":
            try:
                # 获取modem列表
                modem_list = subprocess.run(
                    ["mmcli", "-L"], capture_output=True, text=True
                )
                modem_match = re.search(r"/org/freedesktop/ModemManager1/Modem/(\d+)", modem_list.stdout)
                if modem_match:
                    modem_id = modem_match.group(1)
                    # 获取信号强度
                    signal_result = subprocess.run(
                        ["mmcli", "-m", modem_id, "--signal-get"], capture_output=True, text=True
                    )
                    signal_match = re.search(r"Signal strength\s+:\s+(\d+)%", signal_result.stdout)
                    if signal_match:
                        # 转换为dBm (0-100% -> -113 to -51 dBm)
                        signal_percent = int(signal_match.group(1))
                        return -113 + (signal_percent * 62) // 100
            except:
                pass
                
        return None
        
    def connect_modem(self):
        """连接4G模块"""
        if self.modem_connected:
            return True
            
        try:
            # 获取modem列表
            modem_list = subprocess.run(
                ["mmcli", "-L"], capture_output=True, text=True
            )
            modem_match = re.search(r"/org/freedesktop/ModemManager1/Modem/(\d+)", modem_list.stdout)
            if not modem_match:
                return False
                
            modem_id = modem_match.group(1)
            
            # 启用modem
            subprocess.run(["mmcli", "-m", modem_id, "--enable"], check=True)
            time.sleep(2)
            
            # 获取可用的网络
            networks = subprocess.run(
                ["mmcli", "-m", modem_id, "--3gpp-scan"], capture_output=True, text=True
            )
            
            # 连接到默认网络
            subprocess.run(
                ["mmcli", "-m", modem_id, "--simple-connect=apn=internet"], 
                check=True
            )
            
            time.sleep(5)
            return True
            
        except Exception as e:
            print(f"Error connecting modem: {e}")
            return False
            
    def switch_to_4g(self):
        """切换到4G网络"""
        try:
            # 禁用WiFi
            subprocess.run(["nmcli", "radio", "wifi", "off"], check=True)
            
            # 连接4G
            return self.connect_modem()
            
        except Exception as e:
            print(f"Error switching to 4G: {e}")
            return False
            
    def switch_to_wifi(self):
        """切换到WiFi网络"""
        try:
            # 启用WiFi
            subprocess.run(["nmcli", "radio", "wifi", "on"], check=True)
            time.sleep(2)
            
            # 连接到已知WiFi
            # 这里需要根据实际WiFi配置修改
            subprocess.run(
                ["nmcli", "device", "wifi", "connect", "YourWiFiSSID", "password", "YourWiFiPassword"],
                check=True
            )
            
            return True
            
        except Exception as e:
            print(f"Error switching to WiFi: {e}")
            return False
```

```python
# error_handler.py
import time
import logging
from enum import Enum

class ErrorLevel(Enum):
    INFO = 1
    WARNING = 2
    ERROR = 3
    CRITICAL = 4

class ErrorHandler:
    def __init__(self, robot_controller):
        self.robot_controller = robot_controller
        self.error_history = []
        self.max_retries = {
            ErrorLevel.INFO: 5,
            ErrorLevel.WARNING: 3,
            ErrorLevel.ERROR: 2,
            ErrorLevel.CRITICAL: 1
        }
        
        # 配置日志
        logging.basicConfig(
            filename='robot_errors.log',
            level=logging.ERROR,
            format='%(asctime)s - %(levelname)s - %(message)s'
        )
        
    def handle_error(self, error_type, error_message, level=ErrorLevel.ERROR, retryable=True):
        """处理错误"""
        error_time = time.time()
        error_entry = {
            "timestamp": error_time,
            "error_type": error_type,
            "message": error_message,
            "level": level.value,
            "retryable": retryable,
            "retries": 0
        }
        
        self.error_history.append(error_entry)
        self.error_history = self.error_history[-100:]  # 保留最近100个错误
        
        # 记录日志
        log_message = f"{error_type}: {error_message}"
        if level == ErrorLevel.INFO:
            logging.info(log_message)
        elif level == ErrorLevel.WARNING:
            logging.warning(log_message)
        elif level == ErrorLevel.ERROR:
            logging.error(log_message)
        elif level == ErrorLevel.CRITICAL:
            logging.critical(log_message)
            
        # 发布错误事件
        self.robot_controller.mqtt_client.publish_event({
            "event_type": "error",
            "error_type": error_type,
            "message": error_message,
            "level": level.name,
            "timestamp": int(error_time)
        })
        
        # 尝试恢复
        if retryable:
            return self._attempt_recovery(error_entry)
        else:
            self._handle_non_retryable_error(error_entry)
            return False
            
    def _attempt_recovery(self, error_entry):
        """尝试错误恢复"""
        max_retries = self.max_retries.get(error_entry["level"], 3)
        
        if error_entry["retries"] >= max_retries:
            self._handle_unrecoverable_error(error_entry)
            return False
            
        error_entry["retries"] += 1
        
        # 根据错误类型执行恢复策略
        recovery_strategies = {
            "mqtt_connection_failed": self._recover_mqtt_connection,
            "video_stream_failed": self._recover_video_stream,
            "motion_control_failed": self._recover_motion_control,
            "network_disconnected": self._recover_network,
            "sensor_data_failed": self._recover_sensors
        }
        
        recovery_func = recovery_strategies.get(error_entry["error_type"])
        if recovery_func:
            try:
                success = recovery_func()
                if success:
                    self.robot_controller.mqtt_client.publish_event({
                        "event_type": "error_recovered",
                        "error_type": error_entry["error_type"],
                        "retries": error_entry["retries"],
                        "timestamp": int(time.time())
                    })
                    return True
            except Exception as e:
                print(f"Recovery failed for {error_entry['error_type']}: {e}")
                
        # 等待后重试
        retry_delay = min(30, 2 **error_entry["retries"])  # 指数退避
        time.sleep(retry_delay)
        return False
        
    def _recover_mqtt_connection(self):
        """恢复MQTT连接"""
        try:
            self.robot_controller.mqtt_client.disconnect()
            time.sleep(2)
            self.robot_controller.mqtt_client.connect()
            return True
        except:
            return False
            
    def _recover_video_stream(self):
        """恢复视频流"""
        try:
            self.robot_controller.video_streamer.stop_stream()
            time.sleep(2)
            self.robot_controller.video_streamer.start_stream()
            return True
        except:
            return False
            
    def _recover_motion_control(self):
        """恢复运动控制"""
        try:
            self.robot_controller.motion_controller.reset()
            time.sleep(2)
            return self.robot_controller.motion_controller.test_connection()
        except:
            return False
            
    def _recover_network(self):
        """恢复网络连接"""
        try:
            network_status = self.robot_controller.device_manager.network_manager.check_network_status()
            
            if not network_status["wifi_connected"] and not network_status["modem_connected"]:
                # 尝试连接4G
                return self.robot_controller.device_manager.network_manager.connect_modem()
            elif network_status["current_interface"] == "wifi" and network_status["signal_strength"] < -85:
                # WiFi信号弱，切换到4G
                return self.robot_controller.device_manager.network_manager.switch_to_4g()
            else:
                return True
        except:
            return False
            
    def _recover_sensors(self):
        """恢复传感器数据"""
        try:
            self.robot_controller.perception.reset_sensors()
            time.sleep(2)
            return True
        except:
            return False
            
    def _handle_unrecoverable_error(self, error_entry):
        """处理无法恢复的错误"""
        print(f"Unrecoverable error: {error_entry['error_type']} - {error_entry['message']}")
        
        # 根据错误级别执行不同的应急措施
        if error_entry["level"] == ErrorLevel.CRITICAL.value:
            # 紧急停止
            self.robot_controller.motion_controller.emergency_stop()
            self.robot_controller.mqtt_client.publish_event({
                "event_type": "emergency_stop",
                "reason": f"Unrecoverable error: {error_entry['error_type']}",
                "timestamp": int(time.time())
            })
            
    def _handle_non_retryable_error(self, error_entry):
        """处理不可重试的错误"""
        print(f"Non-retryable error: {error_entry['error_type']} - {error_entry['message']}")
        
        # 记录并通知，但不尝试恢复
        self.robot_controller.mqtt_client.publish_event({
            "event_type": "non_retryable_error",
            "error_type": error_entry["error_type"],
            "message": error_entry["message"],
            "timestamp": int(time.time())
        })
```
